% Page-breaking and typography rules for the Phase 0 report.
\usepackage{needspace}
\usepackage{etoolbox}
\usepackage{placeins}
\usepackage{microtype}
\usepackage{fancyhdr}
\usepackage{caption}

% Avoid single lines of a paragraph or list item at page boundaries.
\clubpenalty=10000
\widowpenalty=10000
\displaywidowpenalty=10000
\brokenpenalty=10000
\interfootnotelinepenalty=10000
\raggedbottom

% Keep headings and short list items with enough following content.
\pretocmd{\section}{\FloatBarrier}{}{}
\pretocmd{\subsection}{\FloatBarrier\Needspace{10\baselineskip}}{}{}
\pretocmd{\subsubsection}{\Needspace{7\baselineskip}}{}{}
\pretocmd{\item}{\Needspace{4\baselineskip}}{}{}

% Keep figures close to their discussion while allowing sensible page placement.
\renewcommand{\topfraction}{0.90}
\renewcommand{\bottomfraction}{0.80}
\renewcommand{\textfraction}{0.08}
\renewcommand{\floatpagefraction}{0.80}
\setlength{\textfloatsep}{14pt plus 2pt minus 4pt}
\captionsetup{font=small,labelfont=bf}

% Clean, consistent page furniture.
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small UAV Remaining Useful Life Estimation}
\fancyhead[R]{\small Phase 0 Data Analysis}
\fancyfoot[C]{\thepage}
\setlength{\headheight}{14pt}

% Use a dedicated title page instead of placing the title below the contents.
\makeatletter
\renewcommand{\maketitle}{%
  \begin{titlepage}
    \thispagestyle{empty}
    \centering
    \vspace*{0.28\textheight}
    {\Huge\bfseries \@title\par}
    \vspace{1.25em}
    {\Large Phase 0 Data Analysis\par}
    \vfill
  \end{titlepage}%
}
\makeatother
